Modules

One import line, any source.

A module is a folder: the .ag source, its C, C++ and Rust companions, and its resources. An import names a C header, a sibling module, a git commit or a url; each is built with its own build system.

Structs

Value types with C's layout.

A struct is an inlay value: passed, returned and read out of arrays, vectors, pointers and members by value.

struct v2                           # struct is an inlay value type
    x: f32
    y: f32

    operator + [ b: v2 ] -> v2 [ x: x + b.x, y: y + b.y ]
    operator - [ b: v2 ] -> v2 [ x: x - b.x, y: y - b.y ]
    operator * [ k: f32 ] -> v2 [ x: x * k,   y: y * k   ]
    operator left * [ k: f32 ] -> v2 [ x: x * k, y: y * k ]

    func dot [ b: v2 ] -> f32
        return x * b.x + y * b.y

    cast -> string [ 'v2-{x}-{y}' ] # conversion member

struct span2                        # same size as v2: bitcasts
    lo: f32
    hi: f32

    func width [] -> f32
        return hi - lo

Classes

Polymorphism you opt into.

Calls are direct by default; * asks for dynamic dispatch. Members carry their access, meta and change blocks.

class Counter                       # class is heap, ref-counted
    public  count  : i64            # readable outside, written in
    intern  hidden : i64            # module-private
    mutable label  : string         # writable by outsiders

    static  made   : i64            # one per type, not per object

    func init []                    # runs after props are set
        hidden = 7

    func inc []
        count += 1

    public func value [] -> i64
        return count

    intern func peek_hidden [] -> i64
        return hidden

    func tally [] -> i64            # a static is shared by every instance
        made += 1
        return made

    cast -> string [ 'count-{count}' ]
    cast -> bool   [ count != 0 ]

    operator +  [ n: i64 ] -> i64 [ count + n ]
    operator -  [ n: i64 ] -> i64 [ count - n ]
    operator *  [ n: i64 ] -> i64 [ count * n ]
    operator /  [ n: i64 ] -> i64 [ count / n ]
    operator %  [ n: i64 ] -> i64 [ count % n ]
    operator |  [ n: i64 ] -> i64 [ count | n ]
    operator &  [ n: i64 ] -> i64 [ count & n ]
    operator ^  [ n: i64 ] -> i64 [ count ^ n ]
    operator << [ n: i64 ] -> i64 [ count << n ]
    operator >> [ n: i64 ] -> i64 [ count >> n ]

References and memory

Counted, with the cost visible.

Objects are reference counted; ref, @ and lambda contexts show where a hold is taken and where it is given back.

intern payload_dropped : i64

10MB rides in the lambda's context; it must go when the lambda does

features.ag
intern payload_dropped : i64

class Payload
    public data : vec i8 [ 10485760 ]   # ten megabytes onboard

    func touch [] -> i64
        for [ k: i64 0, k < 10485760, k += 4096 ]
            data[ k ] = 7               # make the pages resident
        return 1

    func dealloc []
        payload_dropped += 1

func hold_big [ i: i64 :: big: Payload ] -> i64
    return big.touch[] + i

func make_and_drop [] -> i64
    p  : Payload [ ]                # born at refs 0, on the auto-free list
    fn : lambda hold_big[ p ]       # the context holds it: refs 1
    return fn[ 1 ]

func plain_scope [] -> i64          # same payload, no lambda involved
    p : Payload [ ]
    return p.touch[] + 1

func word_char [ c: i8 ] -> bool

func digit_char [ c: i8 ] -> bool

func html_char [ out: string, c: i8 ]
    if [ c == '<' ]
        out.append[ '&lt;' ]
    el [ c == '>' ]
        out.append[ '&gt;' ]
    el [ c == '&' ]
        out.append[ '&amp;' ]
    el
        out.push[ u32[ c ] ]

func html_text [ out: string, s: string ]
    for [ i: i64 0, i < s.count, i += 1 ]
        html_char[ out, s.chars[ i ] ]

func word_class [ w: string ] -> string
    keywords : ' func class struct enum scalar import export expect return if el else for in while switch case default ifdef ifndef lambda new ref local intern public context extend using alias persist async sync try catch finally break continue asm with as cast construct app true false null abstract static inlay reverse '
    types    : ' i8 i16 i32 i64 u8 u16 u32 u64 f32 f64 bool string path vec map object cstr none any unichar sz num void Au Au_t symbol hook handle '
    if [ keywords.index_of[ ' {w} ' ] >= 0 ] return 'k'
    if [ types.index_of[ ' {w} ' ] >= 0 ] return 't'
    if [ digit_char[ w.chars[ 0 ] ] ] return 'n'

    if [ c >= 'A' && c <= 'Z' ] return 't'

func code_line [ out: string, s: string ]
    n : i64 [ s.count ]
    i : i64 [ 0 ]
    while [ i < n ]
        c : i8 [ s.chars[ i ] ]
        if [ c == '#' ]
            out.append[ '<span class="c">' ]
            html_text[ out, s.mid[ i, n - i ] ]
            out.append[ '</span>' ]
            i = n
        el [ c == '\'' ]
            from : i64 [ i ]
            i += 1
            while [ i < n && s.chars[ i ] != '\'' ]
                if [ s.chars[ i ] == '\\' ]
                    i += 1
                i += 1
            i += 1
            out.append[ '<span class="s">' ]
            html_text[ out, s.mid[ from, i - from ] ]
            out.append[ '</span>' ]
        el [ word_char[ c ] ]
            from : i64 [ i ]
            while [ i < n && (word_char[ s.chars[ i ] ] || s.chars[ i ] == '.') ]
                if [ s.chars[ i ] == '.' && !digit_char[ s.chars[ from ] ] ]
                    break
                i += 1
            w  : s.mid[ from, i - from ]
            wc : word_class[ w ]
            if [ wc.count > 0 ]
                out.append[ '<span class="{wc}">' ]
                html_text[ out, w ]
                out.append[ '</span>' ]
            el
                html_text[ out, w ]
        el
            html_char[ out, c ]
            i += 1
    out.append[ '\n' ]
